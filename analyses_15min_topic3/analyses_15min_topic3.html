5.2 Fold changes of sites and proteins

contr <- c("TSB_vs_SNpulse_at_anc", "TSB_vs_SN_at_anc", "SNpulse_vs_SN_at_anc")

fc_all <- bind_rows(
  read_dea("phospho_STYH", contrasts = contr) |>
    inner_join(early_sites, by = c("locus_tag", "residue", "position")) |>
    mutate(layer = "phosphosite"),
  read_dea("total", contrasts = contr) |>
    filter(locus_tag %in% early_sites$locus_tag) |>
    mutate(layer = "protein")
) |>
  left_join(select(labels, locus_tag, site_label), by = "locus_tag")

fc_all |>
  mutate(comparison = recode(contrast,
                             TSB_vs_SNpulse_at_anc = "TSB vs 15 min",
                             TSB_vs_SN_at_anc      = "TSB vs ~15 h",
                             SNpulse_vs_SN_at_anc  = "15 min vs ~15 h"),
         # A_vs_B: diff > 0 = higher in A, diff < 0 = higher in B
         higher_in = if_else(diff > 0,
                             str_extract(contrast, "^[^_]+"),
                             str_match(contrast, "_vs_([^_]+)")[, 2]),
         fold    = round(2^abs(diff), 1),
         changed = FDR < 0.05 & abs(diff) > 1,
         diff    = round(diff, 2),
         FDR     = signif(FDR, 2)) |>
  arrange(site_label, layer, factor(contrast, levels = contr)) |>
  select(site_label, layer, comparison, diff, fold, higher_in, FDR, changed,
         estimate_type) |>
  knitr::kable()

fc <- fc_all |>
  filter(contrast %in% c("TSB_vs_SNpulse_at_anc", "TSB_vs_SN_at_anc")) |>
  mutate(time = factor(if_else(contrast == "TSB_vs_SNpulse_at_anc", "15 min", "~15 h"),
                       levels = c("15 min", "~15 h")),
         # flip the sign: TSB_vs_X has diff < 0 = higher in PASN
         log2fc  = -diff, lo = -conf.high, hi = -conf.low,

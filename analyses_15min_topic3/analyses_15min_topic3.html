3. Data used

library(dplyr)

phospho_dea <- read_dea("phospho_STYH")

early_sites <- tibble(
  locus_tag = c("B7H15_01690", "B7H15_05655", "B7H15_14290"),
  residue = c("S", "T", "S"),
  position = c(33, 21, 194)
)

phospho_comparison <- phospho_dea |>
  inner_join(
    early_sites,
    by = c("locus_tag", "residue", "position")
  ) |>
  filter(
    contrast %in% c(
      "TSB_vs_SNpulse_at_anc",
      "TSB_vs_SN_at_anc"
    )
  ) |>
  select(
    locus_tag, product, residue, position,
    contrast, diff, FDR, estimate_type
  ) |>
  arrange(locus_tag, contrast)

  dplyr::select(locus_tag, residue, position, contrast, diff, FDR) |>
